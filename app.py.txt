@app.route('/consulta', methods=['GET', 'POST'])
def formulario():
    resultado = ""
    if request.method == 'POST':
        nombre = request.form.get('nombre', 'Cliente')
        duda = request.form.get('duda', 'Consulta')
        
        texto = urllib.parse.quote(f"Olá! Meu nome é {nombre}. Gostaria de agendar: {duda}")
        link_ws = f"https://wa.me/{NUMERO_WHATSAPP}?text={texto}"
        
        resultado = f'''
            <div style="background:#e8f5e9; padding:15px; border-radius:10px; text-align:center; margin-bottom:20px;">
                <p style="color:#2e7d32;"><b>Dados registrados!</b></p>
                <a href="{link_ws}" class="btn btn-ws" target="_blank">ENVIAR WHATSAPP ✅</a>
            </div>
        '''

    return ESTILO + f'''
        <div class="card">
            <h2>Agendar Horário</h2>
            {resultado}
            <form method="post">
                <input type="text" name="nombre" placeholder="Seu nome" required>
